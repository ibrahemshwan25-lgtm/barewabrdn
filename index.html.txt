        function convertToEnglishDigits(str) {
            if (str === null || str === undefined) return '';
            const arabic = '٠١٢٣٤٥٦٧٨٩';
            const persian = '۰۱۲۳۴۵۶۷۸۹';
            return String(str).replace(/[٠-٩۰-۹]/g, ch => {
                const i1 = arabic.indexOf(ch);
                if (i1 !== -1) return String(i1);
                const i2 = persian.indexOf(ch);
                if (i2 !== -1) return String(i2);
                return ch;
            });
        }

                const p = val.split('.');
                if (p.length > 2) val = p[0] + '.' + p.slice(1).join('');
            } else {
                val = val.replace(/[^0-9]/g, '');
            }
            if (inputEl.value !== val) inputEl.value = val;

        function renderWarehouse() {
            let totalMem = 0, recMem = 0;
            appState.families.forEach(f => {
                const m = Number(convertToEnglishDigits(f.members) || 0);
                totalMem += m;
                if (f.receiveStatus === 'وەریگرت') recMem += m;
            });

            document.getElementById('whTargetPeople').textContent = formatNumber(appState.settings.targetPeople);
            document.getElementById('whReceivedPeople').textContent = formatNumber(recMem);
            document.getElementById('whRemainingPeople').textContent = formatNumber(Math.max(0, appState.settings.targetPeople - totalMem));

            const items = [
                { name: 'شەکر', unit: 'کیلۆ', ratio: Number(appState.settings.rations.sugar) },
                { name: 'ئارد (هارد)', unit: 'کیلۆ', ratio: Number(appState.settings.rations.flour) },
                { name: 'برنج', unit: 'کیلۆ', ratio: Number(appState.settings.rations.rice) },
                { name: 'زەیت', unit: 'بتڵ', ratio: Number(appState.settings.rations.oil) },
                { name: 'معجون', unit: 'قوتو', ratio: Number(appState.settings.rations.paste) },
                { name: 'نیسک', unit: 'کیلۆ', ratio: Number(appState.settings.rations.lentils) },
                { name: 'ئارینۆک', unit: 'کیلۆ', ratio: Number(appState.settings.rations.groats) },
            ];

            let html = '';
            items.forEach((item, idx) => {
                const totalInWH = Number(appState.settings.targetPeople) * item.ratio;
                const distributed = totalMem * item.ratio;
                const remaining = totalInWH - distributed;
                const pct = totalInWH > 0 ? (distributed / totalInWH) : 0;
                let badge = '<span class="px-2.5 py-1 bg-emerald-100 text-emerald-800 rounded-lg text-[11px] font-semibold">بەردەستە</span>';
                if (remaining <= 0) badge = '<span class="px-2.5 py-1 bg-teal-100 text-teal-800 rounded-lg text-[11px] font-semibold">تەواو بوو</span>';

                html += `
                    <tr class="hover:bg-slate-50">
                        <td class="p-3 text-center">${idx + 1}</td>
                        <td class="p-3 font-bold">${item.name}</td>
                        <td class="p-3 text-center">${item.unit}</td>
                        <td class="p-3 text-center font-semibold">${item.ratio}</td>
                        <td class="p-3 text-center font-bold">${totalInWH.toFixed(1)}</td>
                        <td class="p-3 text-center font-bold text-emerald-700">${distributed.toFixed(1)}</td>
                        <td class="p-3 text-center font-bold ${remaining < 0 ? 'text-rose-600' : ''}">${remaining.toFixed(1)}</td>
                        <td class="p-3 text-center font-bold">${Math.round(pct * 100)}٪</td>
                        <td class="p-3 text-center">${badge}</td>
                    </tr>
                `;
            });
            document.getElementById('warehouseTableBody').innerHTML = html;
        }

                    showToast('هەڵە لە خوێندنەوەی فایل!', true);
                }
            };
            reader.readAsText(file);

        async function initCloudSystem() {
            populateSettingsForm();
            updateHeaderLabels();
            renderAll();
            if (!firebaseConfig) { updateSyncStatus('error', 'هەور بەردەست نییە'); return; }

            try {
                const app = initializeApp(firebaseConfig);
                auth = getAuth(app);
                db = getFirestore(app);
                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                    await signInWithCustomToken(auth, __initial_auth_token);
                } else {
                    await signInAnonymously(auth);
                }

                onAuthStateChanged(auth, (user) => {
                    if (!user) return;
                    currentUser = user;
                    document.getElementById('displayUserId').textContent = user.uid;
                    subscribeToUserState(user.uid);
                });
            } catch (err) {
                updateSyncStatus('error', 'هەڵە لە هەور');
            }
        }

        function subscribeToUserState(userId) {
            if (!db || !userId) return;
            if (unsubscribeDoc) unsubscribeDoc();
            const ref = doc(db, 'artifacts', appId, 'users', userId, 'rationSystem', 'mainState');
            unsubscribeDoc = onSnapshot(ref, async (snap) => {
                if (snap.exists()) {
                    isApplyingRemoteUpdate = true;
                    const d = snap.data();
                    appState.settings = { ...DEFAULT_SETTINGS, ...(d.settings || {}), rations: { ...DEFAULT_SETTINGS.rations, ...(d.settings?.rations || {}) } };
                    appState.families = Array.isArray(d.families) ? d.families : [];
                    appState.archives = Array.isArray(d.archives) ? d.archives : [];
                    populateSettingsForm();
                    updateHeaderLabels();
                    renderAll();
                    if (!document.getElementById('section-archive').classList.contains('hidden')) renderArchive();
                    isApplyingRemoteUpdate = false;
                    updateSyncStatus('synced', 'لەسەر هەور پارێزراوە');
                } else {
                    await saveState();
                }
            }, (error) => {
                updateSyncStatus('error', 'کێشە لە هاوکاتکردن');
            });
        }

        async function saveState() {
            if (isApplyingRemoteUpdate || !db || !currentUser) return;
            updateSyncStatus('saving', 'پاشەکەوتکردن...');
            try {
                const ref = doc(db, 'artifacts', appId, 'users', currentUser.uid, 'rationSystem', 'mainState');
                await setDoc(ref, { settings: appState.settings, families: appState.families, archives: appState.archives, updatedAt: new Date().toISOString() });
                updateSyncStatus('synced', 'لەسەر هەور پارێزراوە');
            } catch (err) {
                updateSyncStatus('error', 'پاشەکەوت نەکرا');
            }
        }

        window.onload = () => { initCloudSystem(); };
